cmake_minimum_required(VERSION 3.21)
project(SifChannelSyntheticSmoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(sif_channel_smoke main.cpp "${REPO_ROOT}/native/runner/sif_channel.cpp")
target_include_directories(sif_channel_smoke PRIVATE "${REPO_ROOT}/native/runner")
enable_testing()
add_test(NAME directional_sif_channel COMMAND sif_channel_smoke)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/fileio-fixture.bin" "abc123")
add_executable(fileio_rpc_smoke fileio.cpp "${REPO_ROOT}/native/runner/fileio_rpc.cpp")
target_include_directories(fileio_rpc_smoke PRIVATE "${REPO_ROOT}/native/runner")
add_test(NAME readonly_fileio_rpc COMMAND fileio_rpc_smoke "${CMAKE_CURRENT_BINARY_DIR}/fileio-fixture.bin")
